# Memory layout of the original SimDLL's internal structures, recovered from the decompile.
#
# One block per struct:
#   struct <Name> <size in hex>
#     <offset> <type> <field name>   [# note]
# Types: u8 u16 u32 i32 i64 f32 f64 bool ptr (void*), <Name>* (pointer to a struct here),
# vec<T> (a std::vector<T>: three pointers), <Name> (embedded struct), T[N] (array).
# Gaps between fields are filled with padding bytes; unknown fields are named after their
# offset (unk_0x18) until their role is known.
# A line "method <C++ declaration>" inside a block declares a member function in the port's
# C++ struct (ignored for Ghidra).
# sim/layout/gen_layout.py turns this into the Ghidra header, the port's C++ structs
# (port/src/sim_types.h) and the port's offset checks.

struct CellSOA 0x108
  0x00 vec<u16> elementIdx
  0x18 vec<f32> temperature
  0x30 vec<f32> mass
  0x48 vec<u8> properties
  0x60 vec<u8> insulation
  0x78 vec<u8> strengthInfo
  0x90 vec<u8> diseaseIdx        # guess: byte array next to the int diseaseCount
  0xa8 vec<i32> diseaseCount
  0xc0 vec<u8> diseaseInfestationTickCount   # cleared together with the disease (SimData::ClearCell)
  0xd8 vec<f32> diseaseAccumulatedError
  0xf0 vec<f32> unk_0xf0         # zeroed at the end of the constructor

struct SimData 0x748
  0x00 i32 width                 # includes the 1-cell border on each side
  0x04 i32 height
  0x08 i32 numGameCells          # (width - 2) * (height - 2)
  0x0c u32 rngState              # MSVC-style rand(): x = x * 214013 + 2531011, (x >> 16) & 0x7fff
  0x10 i32 flowDirection         # +1 / -1, flipped by every SimBase::UpdateData
  0x14 i32 gasDirection          # +1 / -1, flipped by every gas cell PostProcessCell visits
  0x18 u8 savedOptions           # SimSavedOptions bits: 1 = ENABLE_DIAGONAL_FALLING_SAND
  0x1a u16 unk_0x1a              # an element index unstable solids never fall into
  0x1c u16 vacuumElementIdx      # element written by ClearCell
  0x1e u16 unk_0x1e              # another special element index, skipped by backwall heat exchange
  0x20 CellSOA* cells
  0x28 CellSOA* updatedCells     # guess: written by the update, swapped with cells
  0x30 BackwallSOA* backwall
  0x38 u8* unk_0x38              # per cell; 0xff marks cells outside every world (drained each tick)
  0x40 vec<f32> radiationLight   # how much sky radiation reaches each cell (1.0 at the top)
  0x58 f32* unk_0x58             # per cell; DoSublimation adds the sublimated mass. The game reads it
                                 # as accumulatedFlow; cleared every 15th CopySimDataToGame
  0x60 Vector4f* unk_0x60        # per cell; zeroed when the cell evaporates
  0x70 vec<ActiveRegion> activeRegions
  0x68 u8* unk_0x68              # per-cell flags, DoStateTransition sets the low 5 bits
  0x88 u8* unk_0x88              # per game cell; gates spawning falling liquid (visibility?)
  0x90 u16 displacementDirection # rotates the neighbour order DisplaceGas/DisplaceLiquid try (low 2 bits)
  0x98 SimEvents* events
  0xa0 f32 unk_0xa0              # 0.001
  0xa4 f32 unk_0xa4              # 0.001
  0xa8 u8 unk_0xa8               # when set, ignores unk_0x88
  0xb0 vec<ptr> components       # std::vector<SimComponent*>
  0xc8 ElementConsumer elementConsumer
  0x1e8 ElementEmitter elementEmitter
  0x308 RadiationEmitter radiationEmitter
  0x390 ElementChunk elementChunk
  0x430 BuildingHeatExchange buildingHeatExchange
  0x4d0 BuildingToBuildingHeatExchange buildingToBuildingHeatExchange
  0x558 DiseaseEmitter diseaseEmitter
  0x678 DiseaseConsumer diseaseConsumer
  0x718 bool unk_0x718
  0x719 bool headless
  0x71a bool radiationEnabled
  0x71c f32 unk_0x71c            # 1.1
  0x720 f32 unk_0x720            # 2000.0
  0x724 f32 unk_0x724            # 0.3
  0x728 f32 unk_0x728            # 0.7
  0x72c f32 unk_0x72c            # 0.8
  0x730 vec<WorldRect> worldRects # one per world; read by UpdateExposedToSunPropertyTexture
  method void ClearCell(CellAccessor &cell);
  method uint8_t GetStableTicksRemaining(unsigned long cell);
  method void UpdateComponents(float dt, const Region &region);
  method void UpdateComponentsDataListOnly();
  method void ClearBackwall(int cell);

# Queues frames between the game thread (HandleMessage) and the sim (Begin/ProcessNext/End).
struct SimFrameManager 0xd0
  0x00 u8[40] mutex                # std::mutex
  0x28 SimFrameInfo* current       # receives the messages
  0x30 vec<ptr> freeFrames         # SimFrameInfo*, reused by NewFrame
  0x48 vec<ptr> queuedFrames       # closed by NewGameFrame, waiting for the sim
  0x60 vec<ptr> processingFrames   # taken by BeginFrameProcessing
  0x78 vec<ptr> processedFrames    # handed back by EndFrameProcessing
  0xb0 vec<ActiveRegion> activeRegions  # from the last NewGameFrame, sim coordinates
  0xc8 i32 nextFrame               # index into processingFrames
  method bool HandleMessage(int msg_id, int length, void *data, void *&out);
  method int BeginFrameProcessing();
  method float ProcessNextFrame(SimData &sim);
  method void EndFrameProcessing();
  method void NewFrame();
  method static void ProcessFrame(SimFrameInfo &frame, SimData &sim);
