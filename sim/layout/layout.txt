# Memory layout of the original SimDLL's internal structures, recovered from the decompile.
#
# One block per struct:
#   struct <Name> <size in hex>
#     <offset> <type> <field name>   [# note]
# Types: u8 u16 u32 i32 i64 f32 f64 bool ptr (void*), <Name>* (pointer to a struct here),
# vec<T> (a std::vector<T>: three pointers), <Name> (embedded struct), T[N] (array).
# Gaps between fields are filled with padding bytes; unknown fields are named after their
# offset (unk_0x18) until their role is known.
# A line "method <C++ declaration>" inside a block declares a member function in the port's
# C++ struct (ignored for Ghidra).
# sim/layout/gen_layout.py turns this into the Ghidra header, the port's C++ structs
# (port/src/sim_types.h) and the port's offset checks.

struct CellSOA 0x108
  0x00 vec<u16> elementIdx
  0x18 vec<f32> temperature
  0x30 vec<f32> mass
  0x48 vec<u8> properties
  0x60 vec<u8> insulation
  0x78 vec<u8> strengthInfo
  0x90 vec<u8> diseaseIdx        # guess: byte array next to the int diseaseCount
  0xa8 vec<i32> diseaseCount
  0xc0 vec<u8> diseaseInfestationTickCount   # cleared together with the disease (SimData::ClearCell)
  0xd8 vec<f32> diseaseAccumulatedError
  0xf0 vec<f32> unk_0xf0         # zeroed at the end of the constructor

struct SimData 0x748
  0x00 i32 width                 # includes the 1-cell border on each side
  0x04 i32 height
  0x08 i32 numGameCells          # (width - 2) * (height - 2)
  0x18 u8 unk_0x18
  0x1c u16 vacuumElementIdx      # element written by ClearCell
  0x20 CellSOA* cells
  0x28 CellSOA* updatedCells     # guess: written by the update, swapped with cells
  0x68 u8* unk_0x68              # per-cell flags, DoStateTransition sets the low 5 bits
  0x88 u8* unk_0x88              # per game cell; gates spawning falling liquid (visibility?)
  0x90 u16 unk_0x90
  0x98 SimEvents* events
  0xa0 f32 unk_0xa0              # 0.001
  0xa4 f32 unk_0xa4              # 0.001
  0xa8 u8 unk_0xa8               # when set, ignores unk_0x88
  0xb0 vec<ptr> components       # std::vector<SimComponent*>
  0x718 bool unk_0x718
  0x719 bool headless
  0x71a bool radiationEnabled
  0x71c f32 unk_0x71c            # 1.1
  0x720 f32 unk_0x720            # 2000.0
  0x724 f32 unk_0x724            # 0.3
  0x728 f32 unk_0x728            # 0.7
  0x72c f32 unk_0x72c            # 0.8
  method void ClearCell(CellAccessor &cell);

# A cell of one CellSOA, passed by reference to cell-level helpers.
struct CellAccessor 0x10
  0x0 CellSOA* soa
  0x8 i32 cell

# Events produced during a frame, handed to the game through GameDataUpdate.
# Vector element types and offsets from SimBase::ConsolidateEvents.
struct SimEvents 0x1e0
  0x000 vec<SubstanceChangeInfo> substanceChanges
  0x018 vec<SpawnFallingLiquidInfo> spawnFallingLiquid
  0x030 vec<SpawnOreInfo> spawnOre
  0x048 vec<UnstableCellInfo> unstableCells
  0x060 vec<MeltedInfo> unk_melted_0x60
  0x078 vec<MeltedInfo> unk_melted_0x78
  0x090 vec<MeltedInfo> unk_melted_0x90
  0x0a8 vec<i32> unk_0xa8
  0x0c0 vec<CellMeltedInfo> cellMelted
  0x0d8 vec<CallbackInfo> callbacks
  0x0f0 vec<WorldDamageInfo> worldDamage
  0x108 vec<MassConsumedCallback> massConsumed
  0x120 vec<ConsumedRadiationCallback> radiationConsumed
  0x138 vec<MassEmittedCallback> massEmitted
  0x150 vec<DiseaseConsumptionCallback> diseaseConsumed
  0x168 vec<SpawnFXInfo> spawnFX
  0x180 vec<ComponentStateChangedMessage> componentStateChanged
  0x198 vec<i32> unk_0x198
  0x1b0 vec<BackwallElementChangedInfo> backwallElementChanged
  0x1c8 vec<BackwallShouldTransitionInfo> backwallShouldTransition

# Event records, same layout as the C# structs of the same name in Sim.cs.

  0x4 u16 oldElemIdx
  0x6 u16 newElemIdx

# Element table sent by the game (CreateSimElementsTable, Sim.Element in C#, Pack = 4).
struct Element 0xa4
  0x00 i32 id                              # SimHashes
  0x04 u16 lowTempTransitionIdx
  0x06 u16 highTempTransitionIdx
  0x08 u16 elementsTableIdx
  0x0a u8 state                            # low 2 bits: 0 vacuum, 1 gas, 2 liquid, 3 solid
  0x0b u8 numberOfGradientColors
  0x0c f32 specificHeatCapacity
  0x10 f32 thermalConductivity
  0x14 f32 molarMass
  0x18 f32 solidSurfaceAreaMultiplier
  0x1c f32 liquidSurfaceAreaMultiplier
  0x20 f32 gasSurfaceAreaMultiplier
  0x24 f32 flow
  0x28 f32 viscosity
  0x2c f32 minHorizontalFlow
  0x30 f32 minVerticalFlow
  0x34 f32 maxMass
  0x38 f32 lowTemp
  0x3c f32 highTemp
  0x40 f32 strength
  0x44 i32 lowTempTransitionOreID
  0x48 f32 lowTempTransitionOreMassConversion
  0x4c i32 highTempTransitionOreID
  0x50 f32 highTempTransitionOreMassConversion
  0x54 u16 sublimateIndex
  0x56 u16 convertIndex
  0x58 u32 materialProperties
  0x5c u32 colour
  0x60 u32[6] gradientColours
  0x78 i32 sublimateFX
  0x7c f32 sublimateRate
  0x80 f32 sublimateEfficiency
  0x84 f32 sublimateProbability
  0x88 f32 offGasProbability
  0x8c f32 lightAbsorptionFactor
  0x90 f32 radiationAbsorptionFactor
  0x94 f32 radiationPer1000Mass
  0x98 PhysicsData defaultValues

struct PhysicsData 0xc
  0x0 f32 temperature
  0x4 f32 mass
  0x8 f32 pressure

# Per-element tables derived from Element by CreateElementsTable.
struct ElementTemperatureData 0x38
  0x00 u8 state
  0x02 u16 lowTempTransitionIdx
  0x04 u16 highTempTransitionIdx
  0x06 u16 lowTempTransitionOreIdx         # 0xffff when none
  0x08 u16 highTempTransitionOreIdx
  0x0c f32 specificHeatCapacity
  0x10 f32 thermalConductivity
  0x14 f32 defaultMass
  0x18 f32 minMassForTemperature           # guess: 1.0 for gases, 0.001 otherwise
  0x1c f32 gasSurfaceAreaMultiplier
  0x20 f32 liquidSurfaceAreaMultiplier
  0x24 f32 solidSurfaceAreaMultiplier
  0x28 f32 lowTemp
  0x2c f32 highTemp
  0x30 f32 lowTempTransitionOreMassConversion
  0x34 f32 highTempTransitionOreMassConversion

struct ElementPostProcessData 0x2c
  0x00 u16 sublimateIndex
  0x02 u16 convertIndex
  0x04 u8 state
  0x08 f32 sublimateRate
  0x0c f32 sublimateEfficiency
  0x10 f32 sublimateProbability
  0x14 f32 offGasProbability
  0x18 f32 molarMass
  0x1c f32 strength
  0x20 f32 maxMass
  0x24 f32 minHorizontalFlow
  0x28 i32 sublimateFX

struct ElementLiquidData 0x18
  0x00 u8 state
  0x04 f32 flow
  0x08 f32 viscosity
  0x0c f32 minHorizontalFlow
  0x10 f32 minVerticalFlow
  0x14 f32 maxMass

struct ElementPressureData 0x8
  0x0 u8 state
  0x4 f32 flow

struct ElementPropertyTextureData 0x24
  0x00 u32 colour
  0x04 u32[6] gradientColours
  0x1c u32 materialProperties
  0x20 u8 state
  0x21 u8 numberOfGradientColors

struct ElementLightAbsorptionData 0x8
  0x0 f32 lightAbsorptionFactor
  0x4 f32 inverseDefaultMass               # 1 / defaultValues.mass, +inf for solids

struct ElementRadiationData 0x8
  0x0 f32 radiationAbsorptionFactor
  0x4 f32 radiationPer1000Mass
